## 1. 产品定义与行业痛点

### 1.1 产品定义

**AI 自适应医疗显示系统**是一个面向医疗显示器、诊断工作站、手术显示终端、PACS终端、远程会诊终端的智能显示控制系统。系统通过AI技术自动识别医疗影像类型，动态调整显示参数，实现"不同医疗影像自动匹配最佳显示策略"。

### 1.2 行业痛点分析

| 痛点 | 现状 | 影响 |
|------|------|------|
| **手动切换显示参数** | 医生需根据影像类型手动切换GSDF/Gamma/色彩空间 | 效率低、易出错 |
| **单一显示策略** | 所有影像使用相同参数，无法针对CT/MRI/超声优化 | 诊断精度受限 |
| **色彩一致性差** | 多屏/多设备色彩不一致，缺乏统一管理 | 远程会诊困难 |
| **缺乏智能感知** | 无法感知环境光、用户偏好、影像特征 | 体验欠佳 |
| **维护成本高** | 定期校准依赖人工，设备状态不可知 | 运维负担重 |
| **法规合规难** | DICOM合规、FDA认证成本高 | 准入门槛高 |

## 4. AI影像识别引擎

### 4.1 影像类型识别矩阵

| 影像类型 | 识别特征 | 适用显示策略 | DICOM Tag |
|----------|----------|-------------|-----------|
| **DR/CR** | 高对比度、骨骼结构 | Gamma 2.2、高锐化 | Modality=DX/CR |
| **CT** | 灰度范围-1024~3071HU | 窗口宽/_level切换 | Modality=CT |
| **MRI** | T1/T2加权、多序列 | 序列感知Gamma | Modality=MR |
| **PET-CT** | 融合影像、功能代谢 | 伪彩色映射、SUV映射 | Modality=PT |
| **超声** | 实时、B模式/M模式 | 低延迟Gamma、色彩增强 | Modality=US |
| **内窥镜** | 实时、色彩敏感 | DCI-P3、高饱和度 | Modality=ES |
| **数字病理** | 超大分辨率、WSI | 缩放感知参数、细胞增强 | Modality=SM |
| **术野视频** | 实时、血色敏感 | HDR、低延迟、手术LUT | 来源识别 |

#### 4.2.3 边缘部署优化策略

```python
# 模型优化Pipeline
class ModelOptimizer:
    """边缘模型优化Pipeline"""
    
    def __init__(self, model: torch.nn.Module, platform: str):
        self.model = model
        self.platform = platform
    
    def optimize_for_edge(self) -> OptimizedModel:
        # 1. 结构重参数化 (RepVGG风格)
        model = self.apply_repvgg_conversion()
        
        # 2. 感知量化训练 (QAT)
        model = self.apply_quantization_aware_training()
        
        # 3. NPU算子融合
        model = self.fuse_npu_operators()
        
        # 4. 剪枝 (结构化)
        model = self.structured_pruning(threshold=0.01)
        
        # 5. 平台特定优化
        if self.platform == "rk3588":
            return self.optimize_for_mali_g610(model)
        elif self.platform == "jetson":
            return self.optimize_for_tensorrt(model)
        
        return model
    
    def apply_repvgg_conversion(self):
        """RepVGG结构重参数化"""
        for m in self.model.modules():
            if isinstance(m, nn.BatchNorm2d):
                # 将BN融合到Conv中
                m.fuse_modules()
        return self.model
    
    def apply_quantization_aware_training(self):
        """感知量化训练"""
        qconfig = QConfig(
            activation=HistogramObserver,
            weight=PerChannelMinMaxObserver
        )
        prepare_qat(self.model, qconfig)
        # 微调几个epoch恢复精度
        return self.model
```

### 5.1 GSDF动态切换引擎

#### 5.1.1 GSDF (Grayscale Standard Display Function) 原理

GSDF是DICOM Part 14定义的人眼感知线性化标准，公式为：

```
Jn = a + b·L + c·L² + d·L³ + e·(L^0.5) + f·(L^0.25) + g·ln(L)

其中:
- Jn: 归一化感知亮度
- L: 归一化亮度值 (0.001 ~ 4000 cd/m²)
- a~g: DICOM标准系数
```

#### 5.1.3 LUT预计算与存储

```python
class GSDFLUTGenerator:
    """GSDF LUT预计算器"""
    
    # DICOM GSDF系数 (Jnd = a + bL + cL² + dL³ + eL^0.5 + fL^0.25 + g·ln(L))
    GSDF_COEFFICIENTS = {
        'a': -1.3011877,
        'b': -2.5840191e-2,
        'c': 8.0242636e-2,
        'd': -1.0320229e-1,
        'e': 1.3646699e-2,
        'f': 2.8745620e-2,
        'g': -2.5468404e-3,
    }
    
    # L_min = 0.001, L_max = 4000 cd/m²
    L_MIN = 0.001
    L_MAX = 4000.0
    
    def generate_lut(self, 
                     ambient_luminance: float = 10.0,
                     target_luminance: float = 500.0,
                     bit_depth: int = 12) -> np.ndarray:
        """生成GSDF LUT"""
        
        lut_size = 2 ** bit_depth  # 4096 entries for 12-bit
        lut = np.zeros(lut_size, dtype=np.float32)
        
        # 计算归一化亮度到Jnd的映射
        for i in range(lut_size):
            # 输入值归一化 (假设12bit输入)
            normalized_input = i / (lut_size - 1)
            
            # 映射到实际亮度 (考虑环境光补偿)
            L = self._calculate_luminance(
                normalized_input, 
                ambient_luminance, 
                target_luminance
            )
            
            # GSDF逆变换: 获取目标Jnd对应的输入值
            jnd_input = self._gsdf_forward(normalized_input)
            jnd_target = self._gsdf_forward(L)
            
            # 线性插值得到目标输入值
            lut[i] = self._gsdf_inverse(jnd_target)
        
        # 归一化到[0, 1]
        lut = np.clip(lut, 0, 1)
        
        return lut.astype(np.uint16)  # 12-bit
    
    def _calculate_luminance(self, normalized: float, ambient: float, target: float) -> float:
        """计算考虑环境光补偿的亮度"""
        # 基础亮度
        L_base = normalized * target
        
        # 环境光反射
        L_ambient = ambient * 0.2  # 漫反射系数
        
        return L_base + L_ambient
```

```glsl
// Vulkan Compute Shader: Local Tone Mapping
#version 450

layout(local_size_x = 16, local_size_y = 16) in;
layout(binding = 0) uniform sampler2D input_tex;
layout(binding = 1, rgba16f) uniform writeonly image2D output_img;

// 局部调光参数
layout(push_constant) uniform PushConstants {
    float max_luminance;      // 面板最大亮度 (cd/m²)
    float paper_white;        // 白色亮度
    vec2  resolution;         // 局部调光分区
    float transition_speed;   // 平滑过渡速度
} pc;

void main() {
    ivec2 coord = ivec2(gl_GlobalInvocationID.xy);
    ivec2 size = ivec2(pc.resolution);
    
    // 1. 计算局部区域平均亮度
    vec3 color = texelFetch(input_tex, coord, 0).rgb;
    float luma = dot(color, vec3(0.2126, 0.7152, 0.0722));
    
    // 2. 计算局部调光增益 (LTM - Local Tone Mapping)
    float local_avg = compute_local_average(coord, size);
    float ltm_gain = compute_ltm_gain(luma, local_avg, pc.max_luminance);
    
    // 3. 全局色调映射 (HLG/PQ根据场景选择)
    vec3 tone_mapped = hlg_tonemap(color * ltm_gain);
    
    // 4. 肤色保护 (检测并保护肤色区域)
    if (is_skin_tone(color)) {
        tone_mapped = protect_skin_tone(color, tone_mapped);
    }
    
    imageStore(output_img, coord, vec4(tone_mapped, 1.0));
}

float compute_ltm_gain(float luma, float local_avg, float max_lum) {
    // 基于局部对比度的增益
    float contrast = luma / (local_avg + 0.001);
    
    // 动态范围压缩
    float target = clamp(local_avg * contrast, 0.0, max_lum);
    
    // 增益计算
    return min(1.0, target / (luma + 0.001));
}

vec3 hlg_tonemap(vec3 color) {
    // HLG OETF (Opto-Electronic Transfer Function)
    const float a = 0.17883277;
    const float b = 0.28466892;
    const float c = 0.55991073;
    
    vec3 E = color; // 光强度
    vec3 E normalized = E / pc.max_luminance;
    
    vec3 result = vec3(0.0);
    for (int i = 0; i < 3; i++) {
        if (E_normalized[i] <= 1.0/12.0) {
            result[i] = sqrt(3.0 * E_normalized[i]);
        } else {
            result[i] = a * log(12.0 * E_normalized[i] - b) + c;
        }
    }
    
    return result;
}
```

#### 6.1.2 Vulkan医疗显示Pipeline

```glsl
// Vulkan渲染Pipeline: 医疗影像显示
#version 450

// Descriptor Sets
layout(set=0, binding=0) uniform sampler2D inputTexture;      // DICOM影像
layout(set=0, binding=1) uniform sampler1D gsdfLUT;            // GSDF 1D LUT
layout(set=0, binding=2) uniform sampler3D colorSpaceLUT;       // 3D色彩空间转换
layout(set=0, binding=3) uniform UniformBuffer {
    mat4 colorMatrix;
    vec4 windowLevel;     // x=window, y=level, z=min, w=max
    int enableGSDF;
    int enableLocalEnhance;
    int modality;         // 0=CT, 1=MR, 2=US, 3=DR, etc.
} ubo;

// 输出
layout(location=0) out vec4 fragColor;

// 窗口/层级变换 (CT/MRI)
vec3 applyWindowLevel(vec3 value, vec4 wl) {
    float windowMin = wl.y - wl.x * 0.5;
    float windowMax = wl.y + wl.x * 0.5;
    float normalized = (value.r - windowMin) / (windowMax - windowMin);
    return vec3(clamp(normalized, 0.0, 1.0));
}

// GSDF应用
vec3 applyGSDF(vec3 linearValue) {
    float luminance = dot(linearValue, vec3(0.2126, 0.7152, 0.0722));
    
    // 查找GSDF LUT
    float gsdfValue = texture(gsdfLUT, vec2(luminance, 0.5)).r;
    
    // 返回GSDF校正后的值
    return linearValue * (gsdfValue / (luminance + 0.001));
}

// AI局部增强 (简化版)
vec3 applyLocalEnhance(vec2 uv) {
    // 多尺度锐化
    vec3 blur1 = texture(inputTexture, uv, 1).rgb;  // 大尺度
    vec3 blur2 = texture(inputTexture, uv, 2).rgb;  // 小尺度
    
    vec3 sharp = texture(inputTexture, uv, 0).rgb;
    
    // 边缘保持细节增强
    return sharp + 0.3 * (sharp - 0.5 * (blur1 + blur2));
}

void main() {
    // 纹理坐标
    vec2 texCoord = gl_FragCoord.xy / vec2(gl_FragCoord.w);
    
    // 采样原始影像
    vec4 rawColor = texture(inputTexture, texCoord);
    
    // 窗口/层级变换 (针对CT/MRI)
    vec3 windowed = applyWindowLevel(rawColor.rgb, ubo.windowLevel);
    
    // AI局部增强 (可选)
    vec3 enhanced;
    if (ubo.enableLocalEnhance == 1) {
        enhanced = applyLocalEnhance(texCoord);
    } else {
        enhanced = windowed;
    }
    
    // 色彩空间转换
    vec3 colorManaged = texture(colorSpaceLUT, enhanced).rgb;
    
    // GSDF校正 (医疗模式)
    vec3 finalColor;
    if (ubo.enableGSDF == 1) {
        finalColor = applyGSDF(colorManaged);
    } else {
        finalColor = colorManaged;
    }
    
    // 伽马校正输出
    fragColor = vec4(pow(finalColor, vec3(1.0/2.2)), rawColor.a);
}
```

### 7.3 Vulkan渲染优化策略

```glsl
// Vulkan: 医疗影像渲染优化
#version 450

// Multi-draw indirect for batch rendering
layout(indirectable) uniform;

// 输入纹理 (DMA-BUF导入)
layout(set=0, binding=0) uniform sampler2D inputTextures[];

// 输出
layout(location=0) out vec4 fragColor;

// Push Constants (减少Descriptor访问)
layout(push_constant) uniform PushConstants {
    uint tex_index;        // 当前纹理索引
    uint lut_mode;         // LUT模式 (GSDF/HDR/sRGB)
    float window_width;    // 窗口宽度
    float window_center;   // 窗口中心
} pc;

// 优化: 使用NonTemporal写回
layout(early_fragment_tests) in;

// 优化: 共享公共计算
float linearToGSDF(float L) {
    // DICOM GSDF Jnd计算
    const float a = -1.3011877;
    const float b = -2.5840191e-2;
    const float c = 8.0242636e-2;
    const float d = -1.0320229e-1;
    const float e = 1.3646699e-2;
    const float f = 2.8745620e-2;
    const float g = -2.5468404e-3;
    
    float L_clamped = clamp(L, 0.001, 4000.0);
    float lnL = log(L_clamped);
    
    return a + b*L_clamped + c*L_clamped*L_clamped + 
           d*L_clamped*L_clamped*L_clamped +
           e*sqrt(L_clamped) + f*pow(L_clamped, 0.25) + g*lnL;
}

void main() {

    vec2 uv = gl_FragCoord.xy / vec2(gl_FragCoord.w);
    vec4 color = texture(inputTextures[pc.tex_index], uv);
    
    // 窗口/层级变换
    float windowMin = pc.window_center - pc.window_width * 0.5;
    float windowMax = pc.window_center + pc.window_width * 0.5;
    float normalized = (color.r - windowMin) / (windowMax - windowMin);
    normalized = clamp(normalized, 0.0, 1.0);
    
    // 应用GSDF (可选)
    if (pc.lut_mode == 1) {
        // 快速LUT查找
        normalized = linearToGSDF(normalized * 1000.0) / linearToGSDF(1000.0);
    }
    
    fragColor = vec4(vec3(normalized), color.a);
}
```

```python
class DICOMGSDFCalibrator:
    """DICOM GSDF校准器"""
    
    def __init__(self, display_hardware: DisplayHardware):
        self.display = display_hardware
        self.ambient_sensor = AmbientLightSensor()
    
    def calibrate(self) -> CalibrationReport:
        """执行完整校准流程"""
        
        # 1. 环境光测量
        ambient_lux = self.ambient_sensor.measure()
        ambient_luminance = self._lux_to_luminance(ambient_lux)
        
        # 2. 亮度测量 (使用色度计)
        measurements = self._measure_grayscale_steps()
        
        # 3. GSDF拟合
        measured_jnd = self._luminance_to_jnd(measurements)
        target_jnd = self._generate_target_jnd_curve()
        
        # 4. 计算校正LUT
        correction_lut = self._compute_correction_lut(
            measured_jnd, 
            target_jnd
        )
        
        # 5. 验证校准
        verification = self._verify_calibration(
            correction_lut,
            tolerance=2.0  # ΔE ≤ 2.0
        )
        
        return CalibrationReport(
            ambient_luminance=ambient_luminance,
            measurements=measurements,
            correction_lut=correction_lut,
            verification=verification,
            timestamp=datetime.now(),
            next_calibration_due=self._calculate_next_calibration()
        )
    
    def _luminance_to_jnd(self, luminance: float) -> float:
        """亮度 → Jnd (DICOM GSDF)"""
        # Jnd = a + bL + cL² + dL³ + e√L + fL^(1/4) + g·ln(L)
        c = GSDF_COEFFICIENTS
        
        return (c['a'] + c['b'] * luminance +
                c['c'] * luminance**2 + c['d'] * luminance**3 +
                c['e'] * np.sqrt(luminance) + c['f'] * luminance**0.25 +
                c['g'] * np.log(luminance))
    
    def _compute_correction_lut(self, 
                                measured_jnd: np.ndarray,
                                target_jnd: np.ndarray) -> np.ndarray:
        """计算校正LUT"""
        lut_size = 4096  # 12-bit
        lut = np.zeros(lut_size)
        
        # 反向查找: 从目标Jnd找到需要的输入值
        for i, target in enumerate(target_jnd):
            idx = np.argmin(np.abs(measured_jnd - target))
            lut[i] = idx / (lut_size - 1)
        
        # 平滑插值
        lut = self._smooth_lut(lut)
        
        return lut
```

```python
class ColorConsistencyManager:
    """多屏色彩一致性管理"""
    
    def __init__(self, displays: List[Display]):
        self.displays = displays
        self.calibration_db = CalibrationDatabase()
    
    def synchronize_color(self) -> SyncReport:
        """多屏色彩同步"""
        
        # 1. 获取各屏校准状态
        calib_status = [self._get_calib_status(d) for d in self.displays]
        
        # 2. 找到参考屏 (通常为主诊断屏)
        reference = self._select_reference_display(calib_status)
        
        # 3. 计算跨屏校正矩阵
        correction_matrices = {}
        for display in self.displays:
            if display != reference:
                correction_matrices[display.id] = (
                    self._compute_correction_matrix(
                        reference.profile,
                        display.profile
                    )
                )
        
        # 4. 应用校正
        for display_id, matrix in correction_matrices.items():
            self._apply_correction(display_id, matrix)
        
        # 5. 验证同步结果
        return self._verify_synchronization(reference)
    
    def _compute_correction_matrix(self, 
                                   source: ColorProfile, 
                                   target: ColorProfile) -> np.ndarray:
        """计算色彩校正矩阵 (3x3 + 偏移)"""
        # 源到XYZ
        source_to_xyz = source.rgb_to_xyz_matrix
        
        # XYZ到目标RGB
        xyz_to_target = target.xyz_to_rgb_matrix
        
        # 组合变换
        correction = np.linalg.inv(source_to_xyz) @ xyz_to_target
        
        return correction
```

```python
class FrameSyncProtocol:
    """帧同步协议实现"""
    
    SYNC_MODES = {
        "GENLOCK": {

            "tolerance_us": 1,  # 微秒级
            "requires": ["硬件同步线"]
        },

            "description": "软件时间戳同步",
            "tolerance_us": 100,  # 100微秒
            "requires": ["网络PTP同步"]
        },
        "VRR": {
            "description": "可变刷新率同步",
            "tolerance_ms": 3,  # 帧级
            "requires": ["VRR显示器"]
        }
    }
    
    def __init__(self, displays: List[Display]):
        self.displays = displays
        self.sync_mode = self._auto_detect_sync_mode()
    
    def sync_frame(self, frame_data: FrameData) -> SyncResult:
        """执行帧同步"""
        
        if self.sync_mode == "GENLOCK":
            return self._genlock_sync(frame_data)
        elif self.sync_mode == "G席":
            return self._software_sync(frame_data)
        elif self.sync_mode == "VRR":
            return self._vrr_sync(frame_data)
    
    def _genlock_sync(self, frame: FrameData) -> SyncResult:
        """硬件同步锁相"""
        # 1. 等待VSYNC信号
        vsync_time = self.wait_vsync(self.displays[0])
        
        # 2. 计算帧延迟
        frame_offset = frame.timestamp - vsync_time
        
        # 3. 调整各屏时序
        for display in self.displays[1:]:
            self.adjust_timing(display, frame_offset)
        
        # 4. 发送帧
        for display in self.displays:
            display.send_frame(frame)
        
        return SyncResult(
            mode="GENLOCK",
            jitter_ns=measure_jitter(self.displays),
            skew_ns=measure_skew(self.displays)
        )
```

```python
class FederatedLearningCoordinator:
    """联邦学习协调器"""
    
    def __init__(self, cloud_server: str, edge_nodes: List[EdgeNode]):
        self.cloud_server = cloud_server
        self.edge_nodes = edge_nodes
        
        # 差分隐私配置
        self.privacy_config = PrivacyConfig(
            noise_multiplier=1.1,
            max_grad_norm=1.0,
            secure_aggregation=True
        )
    
    async def train_round(self, round_id: int) -> ModelUpdate:
        """执行一轮联邦学习"""
        
        # 1. 选择参与节点 (基于健康状态和计算能力)
        participants = self._select_participants()
        
        # 2. 下发当前模型
        global_model = await self._get_global_model()
        for node in participants:
            await node.receive_model(global_model)
        
        # 3. 本地训练
        local_updates = await asyncio.gather(*[
            node.train_local(epochs=5)
            for node in participants
        ])
        
        # 4. 差分隐私聚合
        aggregated = self._secure_aggregate(local_updates)
        
        # 5. 更新全局模型
        new_model = self._update_global_model(aggregated)
        
        return new_model
    
    def _secure_aggregate(self, updates: List[ModelUpdate]) -> ModelUpdate:
        """安全聚合 (差分隐私)"""
        
        # 1. 加噪
        noisy_updates = []
        for update in updates:
            noisy = self._add_differential_privacy(update)
            noisy_updates.append(noisy)
        
        # 2. 加权平均
        weights = [u.data_size for u in noisy_updates]
        total_weight = sum(weights)
        
        aggregated = {}
        for key in noisy_updates[0].state_dict.keys():
            aggregated[key] = sum(
                u.state_dict[key] * w / total_weight
                for u, w in zip(noisy_updates, weights)
            )
        
        return ModelUpdate(state_dict=aggregated)
```

```python
class PredictiveMaintenanceEngine:
    """预测性维护引擎"""
    
    def __init__(self):
        self.models = {
            "backlight": BacklightLifeModel(),
            "color_drift": ColorDriftModel(),
            "failure": FailurePredictionModel()
        }
        self.anomaly_detector = TimeSeriesAnomalyDetector()
    
    async def predict_backlight_life(self, metrics: DisplayMetrics) -> LifePrediction:
        """预测背光寿命"""
        
        # 构建时序特征
        features = self._extract_backlight_features(metrics)
        
        # LSTM预测
        predictions = self.models["backlight"].predict(
            features,
            horizon=30 * 24  # 预测30天
        )
        
        # 计算剩余寿命
        remaining_hours = self._calculate_remaining_life(predictions)
        
        # 置信区间
        confidence = self._calculate_confidence_interval(predictions)
        
        return LifePrediction(
            remaining_hours=remaining_hours,
            confidence_95=confidence,
            wear_percentage=metrics.cumulative_hours / self.MAX_BACKLIGHT_HOURS,
            recommended_replacement_date=self._estimate_replacement(remaining_hours)
        )
    
    async def detect_color_drift(self, metrics: DisplayMetrics) -> ColorDriftReport:
        """检测色彩漂移"""
        
        # 测量当前色彩
        current_profile = self._measure_color_profile(metrics)
        
        # 获取基准
        baseline = self.calibration_db.get_baseline(metrics.display_id)
        
        # 计算ΔE
        delta_e = color_difference(current_profile, baseline)
        
        # 预测未来漂移
        drift_rate = self._calculate_drift_rate(metrics)
        predicted_delta_e = delta_e + drift_rate * 30  # 30天后
        
        # 判断是否需要校准
        needs_calibration = (
            delta_e > self.DELTA_E_THRESHOLD or
            predicted_delta_e > self.DELTA_E_THRESHOLD * 1.5
        )
        
        return ColorDriftReport(
            current_delta_e=delta_e,
            predicted_delta_e=predicted_delta_e,
            drift_rate=drift_rate,
            needs_calibration=needs_calibration,
            recommended_date=self._estimate_calibration_date(
                predicted_delta_e,
                drift_rate
            )
        )
    
    async def predict_failure(self, metrics: DisplayMetrics) -> FailurePrediction:
        """预测故障概率"""
        
        # 时序异常检测
        anomalies = self.anomaly_detector.detect(metrics.time_series)
        
        # XGBoost故障预测
        features = self._extract_failure_features(metrics)
        probability = self.models["failure"].predict_proba(features)[1]
        
        # 风险等级
        risk_level = self._classify_risk(probability, anomalies)
        
        # 故障类型预测
        failure_type = self._predict_failure_type(metrics)
        
        return FailurePrediction(
            probability_7d=probability,
            risk_level=risk_level,
            predicted_failure_type=failure_type,
            confidence=self.models["failure"].get_confidence(),
            recommended_action=self._get_action(risk_level, failure_type)
        )
```

```python
class OTAUpdateAgent:
    """OTA更新代理"""
    
    def __init__(self, device_info: DeviceInfo):
        self.device = device_info
        self.update_server = "https://ota.medical-display.com"
        self.current_version = device_info.firmware_version
        self.partition_manager = ABBootstrapManager()
    
    async def check_for_updates(self) -> Optional[UpdateInfo]:
        """检查更新"""
        
        response = await self._http_get(
            f"{self.update_server}/api/v1/device/{self.device.id}/update",
            headers={
                "X-Device-Token": self.device.token,
                "X-Current-Version": self.current_version
            }
        )
        
        if response.status == 200:
            return UpdateInfo(**response.json())
        return None
    
    async def download_and_apply(self, update: UpdateInfo):
        """下载并应用更新"""
        
        # 1. 下载到备用分区
        download_path = await self._download_update(
            update.url,
            progress_callback=self._report_progress
        )
        
        # 2. 验证签名
        if not await self._verify_signature(download_path, update.signature):
            raise SecurityError("Signature verification failed")
        
        # 3. 计算并验证哈希
        expected_hash = update.sha256
        actual_hash = await self._compute_hash(download_path)
        if actual_hash != expected_hash:
            raise IntegrityError("Hash mismatch")
        
        # 4. 应用更新
        await self._apply_update(download_path, update.target_partition)
        
        # 5. 验证并切换
        if await self._verify_installation():
            await self._switch_partition(update.target_partition)
        else:
            raise InstallationError("Post-update verification failed")
    
    async def _apply_update(self, image_path: str, target_partition: str):
        """应用更新到目标分区"""
        
        # 读取分区表
        partition_info = self.partition_manager.get_partition_info(target_partition)
        
        # 擦除目标分区
        self.partition_manager.erase(target_partition)
        
        # 写入新镜像
        async with aiofiles.open(image_path, 'rb') as src:
            offset = partition_info.start_sector * 512
            await self._write_to_partition(target_partition, src, offset)
        
        # 写入更新标记
        self.partition_manager.set_update_marker(
            target_partition,
            source_version=self.current_version,
            target_version=self.update_version,
            timestamp=datetime.now()
        )
    
    async def rollback(self):
        """回滚到上一版本"""
        
        current = self.partition_manager.get_active_partition()
        fallback = self.partition_manager.get_fallback_partition()
        
        # 检查是否有有效的fallback版本
        if not self.partition_manager.has_valid_version(fallback):
            raise RollbackError("No valid fallback version available")
        
        # 切换到fallback分区
        await self.partition_manager.switch_partition(fallback)
        
        # 重启
        await self.system.restart()
```
